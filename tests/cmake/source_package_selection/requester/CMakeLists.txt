# Use normal find_package searches: NO_DEFAULT_PATH would hide the fallback bug.
if(NOT CASE STREQUAL "source_only")
  if(CASE MATCHES "_quiet$")
    set(_mode QUIET)
  else()
    set(_mode REQUIRED)
  endif()
  set(_config CONFIG)
  if(CASE MATCHES "_basic_")
    unset(_config)
  endif()
  if(CASE MATCHES "^parent_reject_")
    find_package(gentest 1.2.2 EXACT ${_config} ${_mode})
  elseif(CASE MATCHES "^parent_component_")
    find_package(gentest 2.0.0 EXACT CONFIG ${_mode} COMPONENTS unsupported)
    if(gentest_FOUND)
      message(FATAL_ERROR "The parent's required-component rejection was ignored")
    endif()
  elseif(CASE MATCHES "^parent_")
    find_package(gentest 2.0.0 EXACT CONFIG ${_mode}
      COMPONENTS runtime OPTIONAL_COMPONENTS optional)
    if(NOT gentest_FOUND)
      message(FATAL_ERROR "The compatible parent package was not found")
    endif()
    file(READ "${CMAKE_BINARY_DIR}/parent-config-loaded" _request)
    if(NOT _request STREQUAL "2.0.0|1|runtime;optional")
      message(FATAL_ERROR "The parent did not receive the version and components: '${_request}'")
    endif()
  else()
    find_package(gentest CONFIG ${_mode})
    if(NOT gentest_FOUND)
      message(FATAL_ERROR "The ordinary package lookup unexpectedly failed")
    endif()
  endif()
endif()

# Check runtime origin independently of the selected helper's bookkeeping.
get_target_property(_imported gentest::gentest_runtime IMPORTED)
if(CASE STREQUAL "installed_only")
  if(NOT _imported)
    message(FATAL_ERROR "Installed-only lookup did not import the runtime")
  endif()
else()
  get_target_property(_runtime_dir gentest::gentest_runtime SOURCE_DIR)
  file(REAL_PATH "${GENTEST_SOURCE_DIR}/src" _expected_runtime_dir)
  file(REAL_PATH "${_runtime_dir}" _runtime_dir)
  if(_imported OR NOT _runtime_dir STREQUAL _expected_runtime_dir)
    message(FATAL_ERROR "The source runtime was replaced: '${_runtime_dir}'")
  endif()
endif()
