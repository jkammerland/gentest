cmake_minimum_required(VERSION 3.31)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(gentest_recording_adapters LANGUAGES CXX)
    enable_testing()
    set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
    find_package(gentest CONFIG REQUIRED)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
endif()

find_package(glaze 8.3.0 EXACT CONFIG REQUIRED)
find_package(cbor_tags 0.24.0 EXACT CONFIG REQUIRED)

foreach(format IN ITEMS json cbor)
    add_executable(gentest_record_${format}_tests main.cpp ${format}.hpp)
    target_link_libraries(gentest_record_${format}_tests PRIVATE gentest::gentest)
    target_compile_definitions(gentest_record_${format}_tests PRIVATE GENTEST_RECORD_${format}=1)
    if(format STREQUAL "json")
        target_compile_features(gentest_record_${format}_tests PRIVATE cxx_std_23)
        target_link_libraries(gentest_record_${format}_tests PRIVATE glaze::glaze)
    else()
        target_compile_features(gentest_record_${format}_tests PRIVATE cxx_std_20)
        target_link_libraries(gentest_record_${format}_tests PRIVATE cbor::tags)
    endif()
    gentest_attach_codegen(gentest_record_${format}_tests)
    add_test(NAME runtime_record_${format} COMMAND ${Python3_EXECUTABLE}
        ${CMAKE_CURRENT_SOURCE_DIR}/verify.py $<TARGET_FILE:gentest_record_${format}_tests>
        ${CMAKE_CURRENT_BINARY_DIR}/${format}-reports ${format})
    set_tests_properties(runtime_record_${format} PROPERTIES LABELS "public-api;runner;reporting;adapters")
    if(TARGET gentest_test_suites)
        add_dependencies(gentest_test_suites gentest_record_${format}_tests)
    endif()
endforeach()

add_executable(gentest_record_json_noexceptions noexceptions.cpp)
target_link_libraries(gentest_record_json_noexceptions PRIVATE gentest::gentest glaze::glaze)
if(MSVC)
    target_compile_options(gentest_record_json_noexceptions PRIVATE /EHs-c-)
    target_compile_definitions(gentest_record_json_noexceptions PRIVATE _HAS_EXCEPTIONS=0)
else()
    target_compile_options(gentest_record_json_noexceptions PRIVATE -fno-exceptions)
endif()
add_test(NAME runtime_record_json_noexceptions COMMAND gentest_record_json_noexceptions)
if(TARGET gentest_test_suites)
    add_dependencies(gentest_test_suites gentest_record_json_noexceptions)
endif()
