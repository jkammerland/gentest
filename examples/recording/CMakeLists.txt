cmake_minimum_required(VERSION 3.31)
project(gentest_recording LANGUAGES CXX)
include(CTest)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(gentest CONFIG REQUIRED)

option(RECORDING_WITH_GLAZE "Record JSON snapshots with Glaze (C++23)" OFF)
option(RECORDING_WITH_CBOR "Record CBOR snapshots with cbor_tags (C++20)" OFF)
add_executable(gentest_recording cases.hpp)
target_compile_features(gentest_recording PRIVATE cxx_std_20)
target_link_libraries(gentest_recording PRIVATE gentest::gentest_main)
if(RECORDING_WITH_GLAZE)
    find_package(glaze 8.3.0 CONFIG REQUIRED)
    target_link_libraries(gentest_recording PRIVATE glaze::glaze)
    target_compile_definitions(gentest_recording PRIVATE RECORDING_WITH_GLAZE=1)
endif()
if(RECORDING_WITH_CBOR)
    find_package(cbor_tags 0.24.0 CONFIG REQUIRED)
    target_link_libraries(gentest_recording PRIVATE cbor::tags)
    target_compile_definitions(gentest_recording PRIVATE RECORDING_WITH_CBOR=1)
endif()
gentest_attach_codegen(gentest_recording OUTPUT_DIR "${CMAKE_CURRENT_BINARY_DIR}/gentest_codegen")
gentest_discover_tests(gentest_recording EXTRA_ARGS
    --bench-epochs=3 --bench-warmup=1 --bench-min-epoch-time-s=0.0001
    --bench-min-total-time-s=0 --bench-max-total-time-s=0.02)
